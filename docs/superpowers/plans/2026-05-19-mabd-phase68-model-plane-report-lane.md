def test_solver_mabd_model_step_matches_cpu_oracle_plane_constraints(self) -> None:
    from mabd_reproduction.experiment_configs import load_spinning_box_config
    from mabd_reproduction.single_body_reports import _oracle_body
    from mabd_reproduction.spinning_box_physics import (
        spinning_box_contact_diagnostics,
        spinning_box_cube_corners,
    )

    root = Path(__file__).resolve().parents[1]
    config = load_spinning_box_config(root / "configs/experiments/single_body_spinning_box.yaml")
    free_result = mabd.solve_cpu_oracle_step(
        q=[config.initial_q],
        qd=[config.initial_qd],
        dt=0.01,
        config=mabd.MABDCPUOracleConfig(bodies=[_oracle_body(config)]),
    )
    free_contact = spinning_box_contact_diagnostics(config, free_result.q[0], free_result.qd[0])
    constraints = [
        mabd.MABDCPUOraclePlaneConstraint(
            body=0,
            rest_point=corner,
            plane_normal=config.contact_surface["plane_normal"],
            plane_offset=float(config.contact_surface["plane_offset"]),
        )
        for corner, signed_distance in zip(
            spinning_box_cube_corners(config),
            free_contact.corner_signed_distances,
            strict=True,
        )
        if float(signed_distance) < 0.0
    ]
    self.assertGreater(len(constraints), 0)

    oracle_result = mabd.solve_cpu_oracle_step(
        q=[config.initial_q],
        qd=[config.initial_qd],
        dt=0.01,
        config=mabd.MABDCPUOracleConfig(
            bodies=[_oracle_body(config)],
            plane_constraints=constraints,
            topology="dense",
        ),
    )
    model_result = _run_spinning_box_solver_mabd_model_step(
        config=config,
        q=config.initial_q,
        qd=config.initial_qd,
        time_step_s=0.01,
        plane_constraints=constraints,
    )

    np.testing.assert_allclose(model_result.q, oracle_result.q[0], atol=1.0e-12)
    np.testing.assert_allclose(model_result.qd, oracle_result.qd[0], atol=1.0e-12)
    self.assertEqual(model_result.plane_constraint_count, len(constraints))
    self.assertEqual(model_result.plane_constraint_accepted_count, len(constraints))
    self.assertEqual(model_result.plane_constraint_skipped_count, 0)
    self.assertAlmostEqual(
        model_result.constraint_residual_norm,
        float(getattr(oracle_result, "constraint_residual_norm", 0.0)),
        places=12,
    )

```python
def _run_spinning_box_solver_mabd_model_step(
    *,
    config: SpinningBoxRunConfig,
    q: np.ndarray,
    qd: np.ndarray,
    time_step_s: float,
    plane_constraints: list[object] | None = None,
) -> SolverMABDModelStepResult:
    plane_constraints = [] if plane_constraints is None else list(plane_constraints)
    builder = newton.ModelBuilder()
    SolverMABD.register_custom_attributes(builder)
    body = _oracle_body(config)
    builder.add_custom(
        "mabd:body",
        q_start=0,
        qd_start=0,
        point_start=0,
        point_count=int(body.precompute.rest_points.shape[0]),
        density=1.0,
        volume=1.0,
        mass_matrix=body.precompute.mass_matrix,
        stiffness_matrix=body.precompute.stiffness_matrix,
        rest_q=body.rest_q,
        rotation_mode=body.rotation_mode,
    )
    for constraint in plane_constraints:
        builder.add_custom(
            "mabd:plane_constraint",
            body=int(constraint.body),
            rest_point=np.asarray(constraint.rest_point, dtype=float),
            plane_normal=np.asarray(constraint.plane_normal, dtype=float),
            plane_offset=float(constraint.plane_offset),
        )
    model = builder.finalize()
    model.q = np.asarray(q, dtype=float).reshape(1, 12).copy()
    model.qd = np.asarray(qd, dtype=float).reshape(1, 12).copy()
    state_in = model.state()
    state_out = model.state()
    state_in.body_q = np.asarray(q, dtype=float).copy()
    state_in.body_qd = np.asarray(qd, dtype=float).copy()
    solver = SolverMABD(model)
    solver.step(state_in, state_out, control=None, contacts=None, dt=time_step_s)
    diagnostics = getattr(solver, "last_cpu_oracle_result", None)
    return SolverMABDModelStepResult(
        q=np.asarray(state_out.body_q, dtype=float).copy(),
        qd=np.asarray(state_out.body_qd, dtype=float).copy(),
        residual_norm=float(getattr(diagnostics, "residual_norm", 0.0)),
        constraint_residual_norm=float(getattr(diagnostics, "constraint_residual_norm", 0.0)),
        plane_constraint_count=int(getattr(diagnostics, "plane_constraint_count", len(plane_constraints))),
        plane_constraint_accepted_count=int(
            getattr(diagnostics, "plane_constraint_accepted_count", 0)
        ),
        plane_constraint_skipped_count=int(
            getattr(diagnostics, "plane_constraint_skipped_count", 0)
        ),
    )
```
